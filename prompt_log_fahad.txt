Input:
How can I filter by status in my get_expenses method query since the status is a variable that can be provided optionally.

Output:
Add a parameter like status_filter=None:                                                                                                                             
                                                                                                                                                                       
    def get_expenses(user, conn, status_filter=None):                                                                                                                  
        query = """                                                                                                                                                    
            SELECT e.id, e.expense_date, c.name AS category, e.amount, e.description, a.status                                                                         
            FROM expenses e                                                                                                                                            
            JOIN categories c ON e.category_id = c.id                                                                                                                  
            JOIN approvals a ON a.expense_id = e.id                                                                                                                    
            WHERE e.user_id = ?                                                                                                                                        
        """                                                                                                                                                            
        params = [user["id"]]                                                                                                                                          
                                                                                                                                                                       
        if status_filter:                                                                                                                                              
            query += " AND a.status = ?"                                                                                                                               
            params.append(status_filter)                                                                                                                               
                                                                                                                                                                       
        query += " ORDER BY e.expense_date DESC"                                                                                                                       
                                                                                                                                                                       
        rows = conn.execute(query, params).fetchall()                                                                                                                  
        return [dict(row) for row in rows]
